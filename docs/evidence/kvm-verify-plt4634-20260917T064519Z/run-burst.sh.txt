#!/bin/bash
. "$(dirname "$0")/common.sh"
W=$VERIFY/burst; sudo rm -rf "$W"; mkdir -p "$W"
hostinfo "$W/host.txt" before
sudo -n env PATH="$PATH" HOME="$HOME" TMPDIR=/tmp TSLS_PROVIDER=firecracker TSLS_SKIP_BUILD=1 \
  bash scripts/e2e/burst.sh > "$W/stdout.txt" 2>&1
echo "burst exit=$?" | tee -a "$W/host.txt"
hostinfo "$W/host.txt" after
sudo chown -R "$(id -u):$(id -g)" "$REPO/docs/evidence" "$W" "$REPO/.kvm/run" 2>/dev/null
leftovers >> "$W/host.txt" 2>&1
tail -30 "$W/stdout.txt"
