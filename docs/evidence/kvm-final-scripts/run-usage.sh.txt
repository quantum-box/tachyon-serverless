#!/bin/bash
# item 3 (PLT-4642 / PLT-4643): usage-e2e + budget-e2e on Firecracker
. "$(dirname "$0")/common.sh"
WHICH="${*:-usage budget}"
W=$VERIFY/usage; mkdir -p "$W"
C="$(cat "$HOSTWT/.kvm/vm/commit.txt" 2>/dev/null)"
for what in $WHICH; do
  E="$W/$what"; sudo rm -rf "$E"; mkdir -p "$E"
  hostinfo "$W/host-$what.txt" before
  t0=$(date -u +%s)
  vmload_sampler "$W/vm-load-$what.tsv"; S=$!
  sudo -n env PATH="$PATH" HOME="$HOME" TMPDIR="$HOME/w" TSLS_PROVIDER=firecracker TSLS_SKIP_BUILD=1 TSLS_COMMIT="$C" \
    TSLS_EVIDENCE_DIR="$E" bash "scripts/usage/$what-e2e.sh" > "$W/stdout-$what.txt" 2>&1
  rc=$?
  kill $S
  echo "$what exit=$rc" | tee -a "$W/host-$what.txt"
  hostinfo "$W/host-$what.txt" after
  macload_window "$t0" "$(date -u +%s)" "$W/mac-load-$what.tsv"
  leftovers >> "$W/host-$what.txt" 2>&1
  fix_owner "$W"
  grep -h -E '^(PASS|FAIL|note)' "$E"/*/summary.txt
done
