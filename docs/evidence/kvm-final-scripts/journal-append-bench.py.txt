#!/usr/bin/env python3
"""Replicates one usage-journal append (crates/application/src/usage/journal.rs::append) on this
host's disk: WAL, synchronous=FULL, BEGIN IMMEDIATE, read state + last chain, insert, update,
commit. Prints per-append latency percentiles (ms) for N appends of a ~1.4 KiB event body.
Same SQLite statements, not the Rust binary: python's sqlite3 wrapper adds microseconds, the
fsync dominates."""
import hashlib, json, os, sqlite3, sys, tempfile, time

n = int(sys.argv[1]) if len(sys.argv) > 1 else 500
directory = sys.argv[2] if len(sys.argv) > 2 else tempfile.gettempdir()
schema = """
CREATE TABLE IF NOT EXISTS function_usage_journal (seq INTEGER PRIMARY KEY AUTOINCREMENT,
 event_id TEXT NOT NULL, tenant_id TEXT NOT NULL, body TEXT NOT NULL, body_bytes INTEGER NOT NULL,
 chain TEXT NOT NULL, appended_at TEXT NOT NULL);
CREATE TABLE IF NOT EXISTS function_usage_journal_state (id INTEGER PRIMARY KEY CHECK (id = 1),
 schema_version INTEGER NOT NULL, cursor_seq INTEGER NOT NULL, cursor_chain TEXT NOT NULL,
 pending_events INTEGER NOT NULL, pending_bytes INTEGER NOT NULL);
INSERT OR IGNORE INTO function_usage_journal_state VALUES (1, 1, 0, 'genesis', 0, 0);
"""
results = {}
for mode in ("FULL", "OFF"):
    path = os.path.join(directory, "journal-bench-%s-%d.db" % (mode, os.getpid()))
    con = sqlite3.connect(path, isolation_level=None)
    con.execute("PRAGMA journal_mode = WAL")
    con.execute("PRAGMA synchronous = %s" % mode)
    con.executescript(schema)
    body = json.dumps({"event_type": "attempt_settled", "pad": "x" * 1400})
    lat = []
    for i in range(n):
        t = time.perf_counter()
        con.execute("BEGIN IMMEDIATE")
        con.execute("SELECT pending_events, pending_bytes, cursor_chain FROM function_usage_journal_state WHERE id = 1").fetchone()
        row = con.execute("SELECT chain FROM function_usage_journal ORDER BY seq DESC LIMIT 1").fetchone()
        prev = row[0] if row else "genesis"
        chain = hashlib.sha256((prev + body).encode()).hexdigest()
        con.execute("INSERT INTO function_usage_journal (event_id, tenant_id, body, body_bytes, chain, appended_at) VALUES (?,?,?,?,?,?)",
                    ("e%d" % i, "tn", body, len(body), chain, "2026-09-18T00:00:00Z"))
        con.execute("UPDATE function_usage_journal_state SET pending_events = pending_events + 1, pending_bytes = pending_bytes + ? WHERE id = 1", (len(body),))
        con.execute("COMMIT")
        lat.append((time.perf_counter() - t) * 1000)
    con.close()
    for suffix in ("", "-wal", "-shm"):
        try:
            os.remove(path + suffix)
        except FileNotFoundError:
            pass
    lat.sort()
    pick = lambda q: round(lat[min(len(lat) - 1, int(q * len(lat)))], 3)
    results["synchronous_" + mode.lower()] = {"n": n, "p50_ms": pick(0.5), "p95_ms": pick(0.95), "p99_ms": pick(0.99), "max_ms": round(lat[-1], 3)}
print(json.dumps({"directory": directory, "body_bytes": len(body), "appends": results}, indent=2))
